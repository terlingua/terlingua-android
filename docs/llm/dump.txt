==> .github/workflows/pr-check.yml <==
name: PR check

on:
  pull_request:
    branches: [main]

permissions:
  contents: read

concurrency:
  group: pr-${{ github.event.pull_request.number }}
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-24.04
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@v7

      - uses: actions/setup-java@v6
        with:
          distribution: temurin
          java-version: 21

      - uses: gradle/actions/setup-gradle@v6
        with:
          cache-provider: basic
          cache-read-only: true

      - name: Install Android SDK packages
        run: yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-37" "build-tools;36.0.0" > /dev/null

      - name: Content compiler tests
        run: ./gradlew -p build-logic test

      - name: Unit and Compose UI tests
        run: ./gradlew :app:testDebugUnitTest

      - name: Build debug APK
        run: ./gradlew :app:assembleDebug

      - name: Verify no network permissions
        run: |
          apk=$(ls app/build/outputs/apk/debug/*.apk | head -n1)
          perms=$("$ANDROID_HOME/build-tools/36.0.0/aapt2" dump permissions "$apk")
          echo "$perms"
          if echo "$perms" | grep -E 'android.permission.(INTERNET|ACCESS_LOCAL_NETWORK|ACCESS_NETWORK_STATE)'; then
            echo "::error::Network permission found in merged manifest"; exit 1
          fi


==> .github/workflows/release.yml <==
name: Release

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: release
  cancel-in-progress: false

jobs:
  release:
    runs-on: ubuntu-24.04
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v7

      - uses: actions/setup-java@v6
        with:
          distribution: temurin
          java-version: 21

      - uses: gradle/actions/setup-gradle@v6
        with:
          cache-provider: basic

      - name: Install Android SDK packages
        run: yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-37" "build-tools;36.0.0" > /dev/null

      - name: Compute version
        id: version
        run: |
          code=${{ github.run_number }}
          name="$(date -u +%Y.%m.%d).${{ github.run_number }}"
          echo "code=$code" >> "$GITHUB_OUTPUT"
          echo "name=$name" >> "$GITHUB_OUTPUT"

      - name: Decode signing key
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
        run: |
          if [ -z "$KEYSTORE_BASE64" ]; then echo "::error::KEYSTORE_BASE64 secret is not set"; exit 1; fi
          echo "$KEYSTORE_BASE64" | base64 --decode > "$RUNNER_TEMP/release.keystore"
          echo "TERLINGUA_KEYSTORE_FILE=$RUNNER_TEMP/release.keystore" >> "$GITHUB_ENV"

      - name: Content compiler tests
        run: ./gradlew -p build-logic test

      - name: Unit and Compose UI tests
        run: ./gradlew :app:testDebugUnitTest

      - name: Build signed universal release APK
        env:
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: >
          ./gradlew :app:assembleRelease
          -Pterlingua.versionCode=${{ steps.version.outputs.code }}
          -Pterlingua.versionName=${{ steps.version.outputs.name }}

      - name: Verify, rename and checksum APK
        id: apk
        run: |
          src=app/build/outputs/apk/release/app-release.apk
          tools="$ANDROID_HOME/build-tools/36.0.0"
          "$tools/apksigner" verify --print-certs "$src"
          if "$tools/aapt2" dump permissions "$src" | grep -E 'android.permission.(INTERNET|ACCESS_LOCAL_NETWORK|ACCESS_NETWORK_STATE)'; then
            echo "::error::Network permission found in release APK"; exit 1
          fi
          out="terlingua-books-${{ steps.version.outputs.name }}.apk"
          cp "$src" "$out"
          sha256sum "$out" > "$out.sha256"
          echo "file=$out" >> "$GITHUB_OUTPUT"

      - name: Publish GitHub Release
        uses: softprops/action-gh-release@v3
        with:
          tag_name: v${{ steps.version.outputs.name }}
          name: Terlingua Books ${{ steps.version.outputs.name }}
          target_commitish: ${{ github.sha }}
          prerelease: false
          draft: false
          make_latest: true
          generate_release_notes: true
          files: |
            ${{ steps.apk.outputs.file }}
            ${{ steps.apk.outputs.file }}.sha256

      - name: Remove signing key
        if: always()
        run: rm -f "$RUNNER_TEMP/release.keystore"

# Terlingua Books (Android)

A free, open-source, 100% offline native Android reader and quiz companion for Terlingua Books,
distributed directly via GitHub Releases and Obtainium — no Google Play Store and no internet
connection required.

Built with **Kotlin**, **Jetpack Compose**, **Material 3 Adaptive**, **Navigation 3**, and
**Room 3 (SQLite + FTS5)** on **AGP 9.5+** and **Gradle 9.9+**. Targets **Android 8.0 (API 26)**
through **Android 17 (API 37)** across candy-bar phones, foldables, tablets, and notebook-style
Android PCs.

## Features

- **100% Offline & Private** — Zero network permissions (`INTERNET` and `ACCESS_LOCAL_NETWORK` are not declared); all books, authors, avatars, code highlighting, and quizzes are compiled into the APK at build time
- **Offline Full-Text Search** — Instant SQLite FTS5 search across all books, chapters, and sections
- **Interactive MCQs** — Single-answer (radio) and multiple-answer (checkbox) quizzes with immediate evaluation, retry, "Show answer", Markdown explanations, and sticky completion state
- **Adaptive Multi-Device UI** — Material 3 `ListDetailPaneScaffold` adapts seamlessly to phones, foldables, tablets, and desktop/Chromebook windows with full keyboard and mouse support
- **Themes** — System, Light, Dark, Sepia, Solarized Light, Solarized Dark, and High Contrast
- **Multisensory Feedback** — Material 3 visual ripples alongside optional haptic feedback and lightweight `SoundPool` audio cues (configurable in Settings and respectful of silent/DND mode)
- **Offline Progress Backup & Sharing** — Export/import portable `.terlingua-progress.json` files (`schemaVersion = 1`) via Android Share Sheet (messaging apps, email, Bluetooth, Quick Share, USB, Syncthing) or Storage Access Framework, plus air-gapped QR code transfer with conflict-free merging
- **Automated Releases** — Every push to `main` builds and signs a universal APK and publishes a full GitHub Release for Obtainium
- **About & Licenses** — In-app screen listing the AGPLv3 license, source link and every bundled component

## Quick Start

Requirements: JDK 21 (or 25) and an Android SDK with platform 37 and build-tools 36.

```bash
./gradlew -p build-logic test       # content compiler tests (YAML/Markdown -> SQLite FTS5)
./gradlew :app:testDebugUnitTest    # unit, integration and Robolectric Compose UI tests
./gradlew :app:assembleDebug        # compiles content/ into the APK automatically
./gradlew :app:installDebug         # install on a connected device
./gradlew testAll                   # both test suites
```

The build works with an empty `content/books/` (the app shows an empty state). Tests never read
`content/`; they use fixtures in `app/src/test/resources/fixtures/`.

## Architecture

- **`build-logic/`** — Gradle plugin `terlingua.content`. A per-variant `compile<Variant>TerlinguaContent`
  task (configuration-cache safe, cacheable) validates `content/`, renders Markdown to a JSON block tree
  with build-time syntax highlighting, writes `terlingua/terlingua-books.db` (with an FTS5 index) and
  copies referenced images. Output is wired in with `variant.sources.assets.addGeneratedSourceDirectory`,
  so nothing generated is committed.
- **`app/`** — single-activity Compose app. `domain/` (models, `BookProgress` schema v1, merge/codec
  logic), `data/` (bundled-SQLite content reader, Room 3 progress store, DataStore settings, SAF/FileProvider
  I/O), `ui/` (Navigation 3 destinations, `ViewModel` + `StateFlow` per screen), `feedback/` (central
  ripple + haptic + `SoundPool` feedback).
- The content database is opened with the **bundled SQLite driver** so FTS5 is available on every API level.

## Keyboard, mouse and large screens

- Tablets, unfolded foldables, Chromebooks and desktop windows show the table of contents and the
  reader side by side (`NavigableListDetailPaneScaffold`); state survives folding and resizing.
- `Esc` goes back, `←`/`→` move between sections in the reader, `Tab` moves focus with visible focus rings,
  and mouse hover/wheel work everywhere. No orientation or aspect-ratio locks are declared.

## Content Format

All book content lives under `content/` and is compiled into a read-only SQLite FTS5 database
inside the APK during the Gradle build.

### Book and chapter manifests

```yaml
# content/books/<book-slug>/book.yml
title: Blazor Basics
author: terlingua-team          # author id (file name in content/authors), or a list
summary: Learn the fundamentals.
date: 2026-01-15                # optional; books dated after the UTC build date are skipped
updated: 2026-03-01             # optional
tags: [dotnet, web]
image: images/books/blazor.png  # optional, relative to content/
draft: false                    # optional

# content/books/<book-slug>/<NN>-<chapter-slug>/chapter.yml
title: Getting Started
summary: Optional chapter summary.
draft: false
```

Images in Markdown (`![alt](../../../images/x.png)` or `/images/x.png`) are bundled when they resolve to a
file under `content/`; remote images are shown as links because the app has no network access.

### Directory Layout

```text
content/
├── authors/
│   └── terlingua-team.yml        # author profile
├── images/
│   └── authors/
│       └── terlingua-team.png    # avatar referenced by `avatar: images/authors/...`
└── books/
    └── blazor-basics/            # book slug (lowercase, digits, single hyphens)
        ├── book.yml              # title, author, summary, date, updated, tags, image, draft
        ├── 01-getting-started/   # NN-chapter-slug; number sets display order
        │   ├── chapter.yml       # title, summary, draft
        │   ├── 01-intro.md       # NN-subchapter-slug.md
        │   └── 02-setup.md
        └── 02-components/
            └── ...
```

Chapters and subchapters are ordered numerically by their prefix (`01`, `05`, `10` gaps are fine).
Drafts (`draft: true`) and future-dated books (`date` after the UTC build date) are skipped.

### Author Manifest (`content/authors/<id>.yml`)

```yaml
name: Terlingua Team
email: hello@terlingua.example
bio: Building free, open-source learning tools.
avatar: images/authors/terlingua-team.png
socials:
  github: terlingua
  mastodon: "@terlingua@mastodon.social"
```

### Subchapter Format (`NN-slug.md`)

```markdown
---
title: What is this concept?
questions:
  - id: runtime                   # optional, defaults to q1, q2…; keep stable once published
    prompt: Where does the code run?
    explanation: Shown after a correct answer or when revealed.   # optional
    options:
      - text: On a remote server
      - text: Locally on your device
        correct: true
  - id: properties
    prompt: Which statements are true?
    type: multiple                # optional: single | multiple
    options:
      - id: offline               # optional, defaults to a, b, c…
        text: It works without internet access
        correct: true
      - text: It requires an account
---
Markdown body…
```

- `type` is inferred when omitted: more than one `correct: true` option implies `multiple`.
- Set `type: multiple` explicitly to show checkboxes even when only one option is correct.
- Prompts, options, and explanations support Markdown (inline code, emphasis, links).
- Reader progress is keyed on question and option IDs—keep IDs stable once published.
- Invalid content (no correct option, fewer than 2 options, unnumbered files, missing `chapter.yml`, malformed YAML, duplicate IDs) fails the Gradle build with a clear error list.
- Quote option text that YAML would parse as booleans, numbers, or null (`"Yes"`, `"50"`), and avoid `yes`, `no`, `true`, `false`, and `null` as IDs.

## Offline Progress & Sharing

- A section is complete once every question has been answered correctly, or—if it has no questions—once it has been opened.
- Progress is stored locally on device using the `BookProgress` v1 schema (`schemaVersion = 1`).
- **Share / Backup:** Export a `.terlingua-progress.json` file via the Android Share Sheet (works over Signal, WhatsApp, Telegram, Email, Bluetooth, Quick Share, Syncthing, or local Files via Storage Access Framework).
- **QR Transfer:** Transfer a single book's progress air-gapped between two devices using a compressed QR code (open-source ZXing + CameraX; no Google Play Services).
- **Conflict-Free Merge:** Imports default to a safe merge (`answeredCorrectly` is sticky via logical OR, earliest `firstVisitedAt` is preserved, and newest `updatedAt` wins for last-visited location), with a preview diff before applying and an optional Replace mode.

## Releases & Obtainium

Every push to `main` runs tests, builds a **single universal signed release `.apk`**, and publishes a
**full GitHub Release** (never pre-release) with a `.sha256` checksum.

### Installing with Obtainium

1. Install [Obtainium](https://github.com/ImranR98/Obtainium).
2. Add a new app with source URL: `https://github.com/terlingua/terlingua-android`
3. Set the APK filter regex (if needed) to `.*\.apk$`.
4. Verify and pin the signing certificate SHA-256 fingerprint on first install using `apksigner` or Obtainium's built-in hash check:
   ```bash
   apksigner verify --print-certs terlingua-books-*.apk
   ```

### Creating a signing key (once)

```bash
keytool -genkeypair -v -keystore release.keystore -alias terlingua -keyalg RSA -keysize 4096 -validity 10000
base64 -w0 release.keystore > release.keystore.base64   # paste into the KEYSTORE_BASE64 secret
```

Keep the keystore safe: Obtainium (and Android) only accept updates signed with the same key.

### CI Signing Secrets

GitHub Actions (`release.yml`) expects these repository secrets:

- `KEYSTORE_BASE64` — Base64-encoded release `.jks`/`.keystore`
- `KEYSTORE_PASSWORD` — Keystore password
- `KEY_ALIAS` — Signing key alias
- `KEY_PASSWORD` — Signing key password

## LLM Context Export

`export.sh` writes `docs/llm/dump.txt`, a snapshot of the repository for LLM context:

- **`## File Tree`** — Indented tree of **all git-tracked files** (including `content/`)
- **`## Path only`** — Binary files outside `content/` and `docs/`
- **`## Files`** — Full text of source, Gradle config, workflows, and `README.md` (`content/` and `docs/` file bodies are excluded)

```bash
bash export.sh                         # whole repository (excluding content/ and docs/ bodies)
bash export.sh -- app/src              # narrow with git pathspecs
bash export.sh -f -o /tmp/book.txt -- content/books/my-book   # explicitly include full text of one book
```

## Feedback settings

Buttons and quiz answers combine Material ripples, haptics (`HapticFeedbackType`, honouring the system
touch-feedback setting) and tiny bundled WAV cues played through `SoundPool`. Sound and haptics can be
toggled independently in Settings; sounds never play in silent, vibrate or Do Not Disturb modes, or while
the app is in the background.

    /** Keeps status/navigation bar icons legible when the in-app theme differs from the system theme. */
    private fun applySystemBars(dark: Boolean) {
        val style = if (dark) SystemBarStyle.dark(Color.TRANSPARENT) else SystemBarStyle.light(Color.TRANSPARENT, Color.TRANSPARENT)
        enableEdgeToEdge(statusBarStyle = style, navigationBarStyle = style)
    }

        val uri: Uri? = when (intent.action) {
            Intent.ACTION_VIEW -> intent.data
            Intent.ACTION_SEND -> IntentCompat.getParcelableExtra(intent, Intent.EXTRA_STREAM, Uri::class.java)
            else -> null
        }
        val text = if (intent.action == Intent.ACTION_SEND) intent.getStringExtra(Intent.EXTRA_TEXT) else null
        if (uri == null && text.isNullOrBlank()) return
        lifecycleScope.launch {
            val content = uri?.let { runCatching { container.files.read(it) }.getOrNull() } ?: text

import android.content.Context
import android.net.Uri
import androidx.core.content.FileProvider
import io.github.terlingua.books.domain.ProgressBackup
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.io.IOException
import java.time.LocalDate
import java.time.ZoneOffset

/** Storage Access Framework and FileProvider I/O for portable progress files. */
class ProgressFiles(private val context: Context) {

    fun suggestedName(): String = "terlingua-${LocalDate.now(ZoneOffset.UTC)}${ProgressBackup.FILE_SUFFIX}"

    suspend fun write(uri: Uri, text: String) = withContext(Dispatchers.IO) {
        val stream = context.contentResolver.openOutputStream(uri, "wt") ?: throw IOException("Cannot open $uri")
        stream.use { it.write(text.toByteArray()) }
    }

    suspend fun read(uri: Uri): String = withContext(Dispatchers.IO) {
        val stream = context.contentResolver.openInputStream(uri) ?: throw IOException("Cannot open $uri")
        stream.use { input ->
            val bytes = input.readNBytesCompat(MAX_BYTES + 1)
            if (bytes.size > MAX_BYTES) throw IOException("File is larger than ${MAX_BYTES / 1_000_000} MB")
            bytes.decodeToString()
        }
    }

    /** Writes a cache copy and returns a content:// URI that can be granted to the share target. */
    suspend fun shareableUri(text: String): Uri = withContext(Dispatchers.IO) {
        val dir = File(context.cacheDir, "exports").apply { mkdirs() }
        dir.listFiles()?.forEach { it.delete() }
        val file = File(dir, suggestedName()).apply { writeText(text) }
        FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
    }

    private fun java.io.InputStream.readNBytesCompat(limit: Int): ByteArray {
        val out = java.io.ByteArrayOutputStream()
        val buffer = ByteArray(8192)
        while (out.size() < limit) {
            val n = read(buffer, 0, minOf(buffer.size, limit - out.size()))
            if (n < 0) break
            out.write(buffer, 0, n)
        }
        return out.toByteArray()
    }

    companion object {
        const val MAX_BYTES = 5_000_000
    }
}

/**
 * Read-only access to the bundled content database. The bundled SQLite driver is used because the
 * platform SQLite on older devices is not guaranteed to ship FTS5.
 */
class ContentDatabase(
    private val context: Context,
    private val dispatcher: CoroutineDispatcher = Dispatchers.IO,
) {
    private val mutex = Mutex()
    private var connection: SQLiteConnection? = null

    suspend fun <T> read(block: (SQLiteConnection) -> T): T = withContext(dispatcher) {
        mutex.withLock { block(connection ?: open().also { connection = it }) }
    }

    private fun open(): SQLiteConnection {
        val dir = File(context.noBackupFilesDir, "content").apply { mkdirs() }
        val db = File(dir, "terlingua-books.db")
        val marker = File(dir, "installed.stamp")
        val stamp = installStamp()
        if (!db.isFile || !marker.isFile || marker.readText() != stamp) {
            val tmp = File(dir, "terlingua-books.db.tmp")
            context.assets.open(ContentSql.ASSET_PATH).use { input -> tmp.outputStream().use { input.copyTo(it) } }
            db.delete()
            check(tmp.renameTo(db)) { "Could not install content database" }
            marker.writeText(stamp)
        }
        return BundledSQLiteDriver().open(db.absolutePath, SQLITE_OPEN_READONLY)
    }

    private fun installStamp(): String {
        val pm = context.packageManager
        val info = if (Build.VERSION.SDK_INT >= 33) {
            pm.getPackageInfo(context.packageName, PackageManager.PackageInfoFlags.of(0))
        } else {
            @Suppress("DEPRECATION")
            pm.getPackageInfo(context.packageName, 0)
        }
        return "${info.lastUpdateTime}:${PackageInfoCompat.getLongVersionCode(info)}"
    }
}

    abstract suspend fun deleteBook(slug: String)

    @Query("DELETE FROM section_progress WHERE bookSlug = :slug")

        replaceSlugs.forEach {

            deleteSections(it)
            deleteBook(it)
        }
        books.forEach { upsertBook(it) }
        upsertSections(sections)
        upsertQuestions(questions)
    }

                questionId = questionId,
                bookSlug = bookSlug,

        ProgressBackup(exportedAt = clock(), books = snapshot().values.sortedBy { it.bookSlug })

        return ImportPreview(
            backup = backup,
            mergeDiffs = incoming.map { ProgressMerger.diff(local[it.bookSlug], it, ImportMode.Merge) },
            replaceDiffs = incoming.map { ProgressMerger.diff(local[it.bookSlug], it, ImportMode.Replace) },
        )
    }

        val local = snapshot()
        val result = backup.books.groupBy { it.bookSlug }.map { (slug, copies) ->
            val incoming = copies.reduce { a, b -> ProgressMerger.merge(a, b) }
            if (mode == ImportMode.Replace) ProgressMerger.merge(null, incoming) else ProgressMerger.merge(local[slug], incoming)
        }
        dao.writeBooks(
            replaceSlugs = if (mode == ImportMode.Replace) result.map { it.bookSlug } else emptyList(),
            books = result.map { BookStateEntity(it.bookSlug, it.lastSectionKey, it.updatedAt) },
            sections = result.flatMap { b -> b.sections.map { SectionProgressEntity(it.key, b.bookSlug, it.firstVisitedAt, it.updatedAt) } },
            questions = result.flatMap { b ->
                b.sections.flatMap { s ->
                    s.questions.map {
                        QuestionProgressEntity(s.key, it.id, b.bookSlug, it.answeredCorrectly, it.lastSelection.joinToString(","), it.selectedAt, it.updatedAt)
                    }
                }
            },
        )
    }

    private suspend fun snapshot(): Map<String, BookProgress> =
        assemble(dao.allBooks(), dao.allSections(), dao.allQuestions())

    private fun assemble(
        books: List<BookStateEntity>,
        sections: List<SectionProgressEntity>,
        questions: List<QuestionProgressEntity>,
    ): Map<String, BookProgress> {
        val questionsBySection = questions.groupBy { it.sectionKey }
        val sectionsByBook = sections.groupBy { it.bookSlug }
        val slugs = (books.map { it.bookSlug } + sectionsByBook.keys).toSortedSet()
        val states = books.associateBy { it.bookSlug }
        return slugs.associateWith { slug ->
            val bookSections = sectionsByBook[slug].orEmpty().sortedBy { it.sectionKey }.map { s ->
                SectionProgress(
                    key = s.sectionKey,
                    firstVisitedAt = s.firstVisitedAt,
                    updatedAt = s.updatedAt,
                    questions = questionsBySection[s.sectionKey].orEmpty().sortedBy { it.questionId }.map { q ->
                        QuestionProgress(
                            id = q.questionId,
                            answeredCorrectly = q.answeredCorrectly,
                            lastSelection = q.lastSelection.split(',').filter { it.isNotEmpty() },
                            selectedAt = q.selectedAt,
                            updatedAt = q.updatedAt,
                        )
                    },
                )
            }
            BookProgress(
                bookSlug = slug,
                lastSectionKey = states[slug]?.lastSectionKey,
                updatedAt = states[slug]?.updatedAt ?: bookSections.maxOfOrNull { it.updatedAt } ?: 0L,
                sections = bookSections,
            )
        }
    }

class ProgressFormatException(message: String, cause: Throwable? = null) : Exception(message, cause)

object ProgressCodec {
    /** Prefix identifying a single-book QR payload: gzip-compressed JSON, Base64 encoded. */
    const val QR_PREFIX = "TLP1:"

    /** Byte-mode capacity of a version 40 QR code at error correction level L. */
    const val QR_MAX_LENGTH = 2953

    private val json = Json {
        ignoreUnknownKeys = true
        encodeDefaults = true
        explicitNulls = false
    }
    private val pretty = Json(json) { prettyPrint = true }

    fun encode(backup: ProgressBackup): String = pretty.encodeToString(ProgressBackup.serializer(), backup)

    fun decode(text: String): ProgressBackup {
        val backup = try {
            json.decodeFromString(ProgressBackup.serializer(), text.trim().removePrefix("﻿"))
        } catch (e: SerializationException) {
            throw ProgressFormatException("Not a Terlingua progress file", e)
        } catch (e: IllegalArgumentException) {
            throw ProgressFormatException("Not a Terlingua progress file", e)
        }
        if (backup.format != ProgressBackup.FORMAT) throw ProgressFormatException("Unknown format '${backup.format}'")
        if (backup.schemaVersion > ProgressBackup.SCHEMA_VERSION) {
            throw ProgressFormatException("This file needs a newer app version (schema ${backup.schemaVersion})")
        }
        return backup
    }

    fun encodeQr(book: BookProgress): String {
        val raw = json.encodeToString(BookProgress.serializer(), book).toByteArray()
        val gz = ByteArrayOutputStream().also { out -> GZIPOutputStream(out).use { it.write(raw) } }.toByteArray()
        return QR_PREFIX + Base64.getEncoder().withoutPadding().encodeToString(gz)
    }

    fun decodeQr(payload: String): BookProgress {
        if (!payload.startsWith(QR_PREFIX)) throw ProgressFormatException("Not a Terlingua progress QR code")
        return try {
            val gz = Base64.getDecoder().decode(payload.removePrefix(QR_PREFIX).trim())
            val raw = GZIPInputStream(ByteArrayInputStream(gz)).use { it.readBytes() }
            json.decodeFromString(BookProgress.serializer(), raw.decodeToString()).also {
                if (it.schemaVersion > ProgressBackup.SCHEMA_VERSION) throw ProgressFormatException("Newer schema ${it.schemaVersion}")
            }
        } catch (e: ProgressFormatException) {
            throw e
        } catch (e: Exception) {
            throw ProgressFormatException("Damaged progress QR code", e)
        }
    }
}

package io.github.terlingua.books.domain

/**
 * Conflict-free merge: `answeredCorrectly` is sticky (logical OR), the earliest `firstVisitedAt`
 * and the latest `updatedAt` win, and the selection with the newest `selectedAt` is kept.
 * The operation is commutative, associative and idempotent.
 */
object ProgressMerger {

    fun merge(local: BookProgress?, incoming: BookProgress): BookProgress {
        if (local == null) return incoming.normalized()
        require(local.bookSlug == incoming.bookSlug) { "Cannot merge different books" }
        val keys = (local.sections.map { it.key } + incoming.sections.map { it.key }).distinct().sorted()
        val newer = when {
            incoming.updatedAt != local.updatedAt -> if (incoming.updatedAt > local.updatedAt) incoming else local
            else -> if (incoming.lastSectionKey.orEmpty() > local.lastSectionKey.orEmpty()) incoming else local
        }
        return BookProgress(
            bookSlug = local.bookSlug,
            lastSectionKey = newer.lastSectionKey ?: listOfNotNull(local.lastSectionKey, incoming.lastSectionKey).maxOrNull(),
            updatedAt = maxOf(local.updatedAt, incoming.updatedAt),
            sections = keys.map { key -> mergeSection(local.section(key), incoming.section(key)) },
        )
    }

    private fun mergeSection(a: SectionProgress?, b: SectionProgress?): SectionProgress {
        if (a == null) return b!!.normalized()
        if (b == null) return a.normalized()
        val ids = (a.questions.map { it.id } + b.questions.map { it.id }).distinct().sorted()
        return SectionProgress(
            key = a.key,
            firstVisitedAt = listOfNotNull(a.firstVisitedAt, b.firstVisitedAt).minOrNull(),
            updatedAt = maxOf(a.updatedAt, b.updatedAt),
            questions = ids.map { id -> mergeQuestion(a.question(id), b.question(id)) },
        )
    }

    private fun mergeQuestion(a: QuestionProgress?, b: QuestionProgress?): QuestionProgress {
        if (a == null) return b!!
        if (b == null) return a
        val latest = when {
            a.selectedAt != b.selectedAt -> if (a.selectedAt > b.selectedAt) a else b
            else -> if (a.lastSelection.joinToString() >= b.lastSelection.joinToString()) a else b
        }
        return QuestionProgress(
            id = a.id,
            answeredCorrectly = a.answeredCorrectly || b.answeredCorrectly,
            lastSelection = latest.lastSelection,
            selectedAt = latest.selectedAt,
            updatedAt = maxOf(a.updatedAt, b.updatedAt),
        )
    }

    private fun BookProgress.normalized() = copy(
        schemaVersion = ProgressBackup.SCHEMA_VERSION,
        sections = sections.sortedBy { it.key }.map { it.normalized() },
    )

    private fun SectionProgress.normalized() = copy(questions = questions.sortedBy { it.id })

    fun diff(local: BookProgress?, incoming: BookProgress, mode: ImportMode = ImportMode.Merge): BookDiff {
        val result = if (mode == ImportMode.Merge) merge(local, incoming) else incoming
        val before = local?.sections?.associateBy { it.key }.orEmpty()
        var visited = 0
        var correct = 0
        var lostCorrect = 0
        var selections = 0
        result.sections.forEach { s ->
            val old = before[s.key]
            if (s.firstVisitedAt != null && old?.firstVisitedAt == null) visited++
            s.questions.forEach { q ->
                val oq = old?.question(q.id)
                if (q.answeredCorrectly && oq?.answeredCorrectly != true) correct++
                if (q.lastSelection != oq?.lastSelection.orEmpty() && q.lastSelection.isNotEmpty()) selections++
            }
        }
        if (mode == ImportMode.Replace) {
            val after = result.sections.associateBy { it.key }
            lostCorrect = before.values.sumOf { s ->
                s.questions.count { q -> q.answeredCorrectly && after[s.key]?.question(q.id)?.answeredCorrectly != true }
            }
        }
        return BookDiff(
            bookSlug = incoming.bookSlug,
            newlyVisitedSections = visited,
            newlyCorrectQuestions = correct,
            changedSelections = selections,
            lostCorrectQuestions = lostCorrect,
            isNewBook = local == null,
        )
    }
}

data class BookDiff(
    val bookSlug: String,
    val newlyVisitedSections: Int,
    val newlyCorrectQuestions: Int,
    val changedSelections: Int,
    val lostCorrectQuestions: Int = 0,
    val isNewBook: Boolean = false,
) {
    val isEmpty: Boolean
        get() = newlyVisitedSections == 0 && newlyCorrectQuestions == 0 && changedSelections == 0 && lostCorrectQuestions == 0
}

package io.github.terlingua.books.feedback

import android.app.NotificationManager
import android.content.Context
import android.media.AudioAttributes
import android.media.AudioManager
import android.media.SoundPool
import androidx.lifecycle.DefaultLifecycleObserver
import androidx.lifecycle.LifecycleOwner
import androidx.lifecycle.ProcessLifecycleOwner
import io.github.terlingua.books.R
import java.util.concurrent.ConcurrentHashMap

enum class FeedbackEvent { Tap, Toggle, Correct, Incorrect }

/**
 * Plays tiny bundled sonification cues. Sounds are only played while the app is in the foreground
 * (Android 17 restricts background audio) and never in silent, vibrate or Do Not Disturb modes.
 */
class FeedbackPlayer(context: Context) : DefaultLifecycleObserver {
    private val audio = context.getSystemService(AudioManager::class.java)
    private val notifications = context.getSystemService(NotificationManager::class.java)
    private val loaded = ConcurrentHashMap.newKeySet<Int>()
    private val pool = SoundPool.Builder()
        .setMaxStreams(2)
        .setAudioAttributes(
            AudioAttributes.Builder()
                .setUsage(AudioAttributes.USAGE_ASSISTANCE_SONIFICATION)
                .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                .build(),
        )
        .build()
        .apply { setOnLoadCompleteListener { _, id, status -> if (status == 0) loaded += id } }

    private val sounds = mapOf(
        FeedbackEvent.Tap to pool.load(context, R.raw.feedback_tap, 1),
        FeedbackEvent.Toggle to pool.load(context, R.raw.feedback_toggle, 1),
        FeedbackEvent.Correct to pool.load(context, R.raw.feedback_correct, 1),
        FeedbackEvent.Incorrect to pool.load(context, R.raw.feedback_incorrect, 1),
    )

    @Volatile private var foreground = false

    init {
        ProcessLifecycleOwner.get().lifecycle.addObserver(this)
    }

    override fun onStart(owner: LifecycleOwner) {
        foreground = true
        pool.autoResume()
    }

    override fun onStop(owner: LifecycleOwner) {
        foreground = false
        pool.autoPause()
    }

    fun play(event: FeedbackEvent) {
        if (!foreground || !audibleNow()) return
        val id = sounds[event]?.takeIf { it in loaded } ?: return
        val volume = if (event == FeedbackEvent.Tap || event == FeedbackEvent.Toggle) 0.35f else 0.6f
        pool.play(id, volume, volume, 1, 0, 1f)
    }

    private fun audibleNow(): Boolean {
        if (audio?.ringerMode != AudioManager.RINGER_MODE_NORMAL) return false
        val filter = notifications?.currentInterruptionFilter ?: NotificationManager.INTERRUPTION_FILTER_ALL
        return filter == NotificationManager.INTERRUPTION_FILTER_ALL || filter == NotificationManager.INTERRUPTION_FILTER_UNKNOWN
    }
}

        get() = progress?.lastSectionKey?.takeIf { key -> toc?.sections?.any { it.key == key } == true }
            ?: toc?.sections?.firstOrNull()?.key

    fun isComplete(ref: SectionRef): Boolean = Completion.isComplete(ref, progress?.section(ref.key))

    fun isSolved(sectionKey: String, questionId: String): Boolean =
        progress?.section(sectionKey)?.question(questionId)?.answeredCorrectly == true

    fun neighbour(key: String, offset: Int): String? {
        val sections = toc?.sections ?: return null
        val index = sections.indexOfFirst { it.key == key }
        return if (index < 0) null else sections.getOrNull(index + offset)?.key
    }
}

data class QuizUiState(val selection: Set<String> = emptySet(), val result: Result = Result.None, val revealed: Boolean = false) {
    enum class Result { None, Correct, Incorrect }

    val locked: Boolean get() = result == Result.Correct || revealed
}

@OptIn(ExperimentalCoroutinesApi::class)
class BookViewModel(
    private val slug: String,
    private val books: BookRepository,
    private val progress: ProgressRepository,
) : ViewModel() {

    private val toc = viewModelScope.async { books.toc(slug) }

    val state: StateFlow<BookUiState> = combine(flow { emit(toc.await()) }, progress.observeBook(slug)) { toc, p ->
        BookUiState(loading = false, toc = toc, progress = p)
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), BookUiState())

    private val selected = MutableStateFlow<String?>(null)

    val section: StateFlow<SectionUiState> = selected.filterNotNull().transformLatest { key ->
        emit(SectionUiState(key, loading = true))
        emit(SectionUiState(key, loading = false, content = books.section(key)))
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), SectionUiState())

    private val _quiz = MutableStateFlow<Map<String, QuizUiState>>(emptyMap())
    val quiz: StateFlow<Map<String, QuizUiState>> = _quiz.asStateFlow()

    fun open(sectionKey: String) {
        if (selected.value == sectionKey) return
        selected.value = sectionKey
        viewModelScope.launch { progress.markVisited(slug, sectionKey) }
    }

    fun toggle(sectionKey: String, question: Question, optionId: String) = updateQuiz(sectionKey, question) { s ->
        if (s.locked) return@updateQuiz s
        val selection = when {
            !question.multiple -> setOf(optionId)
            optionId in s.selection -> s.selection - optionId
            else -> s.selection + optionId
        }
        s.copy(selection = selection, result = QuizUiState.Result.None)
    }

    /** Evaluates immediately and returns whether the answer was correct (used for feedback cues). */
    fun check(sectionKey: String, question: Question): Boolean {
        val current = _quiz.value[quizKey(sectionKey, question)] ?: QuizUiState()
        val correct = QuizEvaluator.isCorrect(question, current.selection)
        updateQuiz(sectionKey, question) {
            it.copy(result = if (correct) QuizUiState.Result.Correct else QuizUiState.Result.Incorrect)
        }
        viewModelScope.launch {
            progress.recordAnswer(slug, sectionKey, question.id, current.selection.sorted(), correct)
        }
        return correct
    }

    fun retry(sectionKey: String, question: Question) = updateQuiz(sectionKey, question) { QuizUiState() }

    fun reveal(sectionKey: String, question: Question) = updateQuiz(sectionKey, question) { it.copy(revealed = true) }

    private inline fun updateQuiz(sectionKey: String, question: Question, crossinline transform: (QuizUiState) -> QuizUiState) {
        val key = quizKey(sectionKey, question)
        _quiz.update { map -> map + (key to transform(map[key] ?: QuizUiState())) }
    }

    companion object {
        fun quizKey(sectionKey: String, question: Question) = "$sectionKey#${question.id}"
    }
}

fun socialUrl(key: String, value: String): String? {

    val handle = v.removePrefix("@")
    return when (key.lowercase()) {
        "github" -> "https://github.com/$handle"
        "gitlab" -> "https://gitlab.com/$handle"
        "codeberg" -> "https://codeberg.org/$handle"

        "bluesky" -> "https://bsky.app/profile/$handle"

        "website", "web", "homepage", "blog" -> "https://$v"
        else -> null
    }

@Composable
fun rememberInlines(content: List<Inline>): AnnotatedString {
    val colors = MaterialTheme.colorScheme
    val code = LocalCodeColors.current
    return remember(content, colors, code) {
        val linkStyles = TextLinkStyles(SpanStyle(color = colors.primary, textDecoration = TextDecoration.Underline))
        val codeStyle = SpanStyle(fontFamily = FontFamily.Monospace, background = code.background, color = code.plain)
        buildAnnotatedString {
            fun appendInlines(items: List<Inline>) {
                items.forEach { item ->
                    when (item) {
                        is Inline.Text -> append(item.v)
                        is Inline.Code -> withStyle(codeStyle) { append(item.v) }
                        is Inline.Emphasis -> withStyle(SpanStyle(fontStyle = FontStyle.Italic)) { appendInlines(item.c) }
                        is Inline.Strong -> withStyle(SpanStyle(fontWeight = FontWeight.SemiBold)) { appendInlines(item.c) }
                        is Inline.Strike -> withStyle(SpanStyle(textDecoration = TextDecoration.LineThrough)) { appendInlines(item.c) }
                        is Inline.Link -> withLink(LinkAnnotation.Url(item.href, linkStyles)) { appendInlines(item.c) }
                        Inline.Break -> append('\n')
                    }
                }
            }
            appendInlines(content)
        }
    }
}

@Composable
private fun TableBlock(table: Block.Table, modifier: Modifier) {
    val columns = maxOf(table.head.size, table.rows.maxOfOrNull { it.size } ?: 0)
    if (columns == 0) return
    val outline = MaterialTheme.colorScheme.outlineVariant
    val shape = RoundedCornerShape(8.dp)
    val cells = buildList {
        repeat(columns) { add(table.head.getOrNull(it).orEmpty() to true) }
        table.rows.forEach { row -> repeat(columns) { add(row.getOrNull(it).orEmpty() to false) } }
    }
    Box(
        modifier
            .fillMaxWidth()
            .clip(shape)
            .border(1.dp, outline, shape)
            .background(MaterialTheme.colorScheme.surfaceContainerLow)
            .horizontalScroll(rememberScrollState()),
    ) {
        GridLayout(columns = columns) {
            cells.forEachIndexed { index, (content, header) ->
                val align = table.align.getOrNull(index % columns)
                InlineText(
                    content,
                    style = (if (header) MaterialTheme.typography.titleSmall else MaterialTheme.typography.bodyMedium).copy(
                        textAlign = when (align) {
                            "c" -> TextAlign.Center
                            "r" -> TextAlign.End
                            else -> TextAlign.Start
                        },
                    ),
                    modifier = Modifier.padding(horizontal = 12.dp, vertical = 8.dp),
                )
            }
        }
    }
}

/** Measures cells (row-major) so each column takes the width of its widest cell, capped for readability. */
@Composable
private fun GridLayout(columns: Int, content: @Composable () -> Unit) {
    Layout(content = content) { measurables, _ ->
        val maxCell = 320.dp.roundToPx()
        val placeables = measurables.map { it.measure(Constraints(maxWidth = maxCell)) }
        val rows = (placeables.size + columns - 1) / columns
        val widths = IntArray(columns) { c -> (0 until rows).maxOf { r -> placeables.getOrNull(r * columns + c)?.width ?: 0 } }
        val heights = IntArray(rows) { r -> (0 until columns).maxOf { c -> placeables.getOrNull(r * columns + c)?.height ?: 0 } }
        layout(widths.sum(), heights.sum()) {
            var y = 0
            for (r in 0 until rows) {
                var x = 0
                for (c in 0 until columns) {
                    placeables.getOrNull(r * columns + c)?.placeRelative(x, y)
                    x += widths[c]
                }
                y += heights[r]
            }
        }
    }
}

private fun highlighted(text: String, color: androidx.compose.ui.graphics.Color): AnnotatedString = buildAnnotatedString {
    var bold = false
    val chunk = StringBuilder()
    fun flush() {
        if (chunk.isEmpty()) return
        if (bold) withStyle(SpanStyle(fontWeight = FontWeight.Bold, color = color)) { append(chunk.toString()) } else append(chunk.toString())
        chunk.clear()
    }
    text.forEach { c ->
        when (c) {
            SearchHit.HIGHLIGHT_START -> { flush(); bold = true }
            SearchHit.HIGHLIGHT_END -> { flush(); bold = false }
            else -> chunk.append(c)
        }
    }
    flush()
}

@OptIn(FlowPreview::class, ExperimentalCoroutinesApi::class)
class SearchViewModel(private val books: BookRepository) : ViewModel() {
    private val _query = MutableStateFlow("")
    val query: StateFlow<String> = _query.asStateFlow()

    val state: StateFlow<SearchUiState> = _query
        .debounce(200)
        .distinctUntilChanged()
        .mapLatest { q ->
            if (q.isBlank()) SearchUiState(q) else SearchUiState(q, runCatching { books.search(q) }.getOrDefault(emptyList()), searched = true)
        }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), SearchUiState())

    fun onQueryChange(value: String) {
        _query.value = value
    }
}

sealed interface SettingsMessage {
    data object Exported : SettingsMessage
    data class Failed(val reason: String) : SettingsMessage
}

class SettingsViewModel(
    private val settingsRepository: SettingsRepository,
    private val progress: ProgressRepository,
    private val files: ProgressFiles,
    private val incomingImports: MutableStateFlow<String?>,
    books: BookRepository,
) : ViewModel() {

    val settings: StateFlow<Settings> = settingsRepository.settings
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), Settings())

    val books: StateFlow<List<BookSummary>> = flow { emit(books.books()) }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), emptyList())

    private val _message = MutableStateFlow<SettingsMessage?>(null)
    val message: StateFlow<SettingsMessage?> = _message.asStateFlow()

    fun suggestedFileName(): String = files.suggestedName()

    fun setTheme(theme: ThemeMode) = launch { settingsRepository.setTheme(theme) }
    fun setSound(enabled: Boolean) = launch { settingsRepository.setSound(enabled) }
    fun setHaptics(enabled: Boolean) = launch { settingsRepository.setHaptics(enabled) }

    fun exportTo(uri: Uri) = launch {
        runCatching { files.write(uri, ProgressCodec.encode(progress.exportAll())) }
            .onSuccess { _message.value = SettingsMessage.Exported }
            .onFailure { _message.value = SettingsMessage.Failed(it.message ?: it::class.simpleName.orEmpty()) }
    }

    suspend fun shareableUri(): Uri = files.shareableUri(ProgressCodec.encode(progress.exportAll()))

    fun importFrom(uri: Uri) = launch {
        runCatching { files.read(uri) }
            .onSuccess { incomingImports.value = it }
            .onFailure { _message.value = SettingsMessage.Failed(it.message ?: it::class.simpleName.orEmpty()) }
    }

    fun consumeMessage() {
        _message.value = null
    }

    private fun launch(block: suspend () -> Unit) {
        viewModelScope.launch { block() }
    }
}

fun TerlinguaTheme(mode: ThemeMode, content: @Composable () -> Unit) {
    val systemDark = isSystemInDarkTheme()
    val context = LocalContext.current
    val dark = when (mode) {
        ThemeMode.System -> systemDark
        ThemeMode.Dark, ThemeMode.SolarizedDark, ThemeMode.HighContrast -> true
        ThemeMode.Light, ThemeMode.Sepia, ThemeMode.SolarizedLight -> false
    }
    val scheme: ColorScheme = when (mode) {
        ThemeMode.System -> when {
            Build.VERSION.SDK_INT >= 31 && systemDark -> dynamicDarkColorScheme(context)
            Build.VERSION.SDK_INT >= 31 -> dynamicLightColorScheme(context)
            systemDark -> DarkScheme
            else -> LightScheme
        }
        ThemeMode.Light -> LightScheme
        ThemeMode.Dark -> DarkScheme
        ThemeMode.Sepia -> SepiaScheme
        ThemeMode.SolarizedLight -> SolarizedLightScheme
        ThemeMode.SolarizedDark -> SolarizedDarkScheme
        ThemeMode.HighContrast -> HighContrastScheme
    }
    val code = when (mode) {
        ThemeMode.SolarizedLight -> SolarizedLightCode
        ThemeMode.SolarizedDark -> SolarizedDarkCode
        ThemeMode.Sepia -> SepiaCode
        ThemeMode.HighContrast -> HighContrastCode
        else -> if (dark) DarkCode else LightCode
    }
    CompositionLocalProvider(
        LocalCodeColors provides code,
        LocalReaderColors provides if (dark) DarkReader else LightReader,
        LocalIsDarkTheme provides dark,
    ) {
        MaterialTheme(colorScheme = scheme, typography = TerlinguaTypography, content = content)
    }
}

sealed interface ImportUiState {
    data object Idle : ImportUiState
    data class Preview(val preview: ImportPreview, val titles: Map<String, String>) : ImportUiState
    data class Error(val message: String) : ImportUiState
    data object Done : ImportUiState
}

/** Reviews progress documents arriving from files, share intents or QR scans before applying them. */
class ImportViewModel(
    private val progress: ProgressRepository,
    private val books: BookRepository,
    private val incoming: MutableStateFlow<String?>,
    private val clock: () -> Long = System::currentTimeMillis,
) : ViewModel() {

    private val _state = MutableStateFlow<ImportUiState>(ImportUiState.Idle)
    val state: StateFlow<ImportUiState> = _state.asStateFlow()

    init {
        viewModelScope.launch {
            incoming.filterNotNull().collect { text ->
                incoming.value = null
                load(text)
            }
        }
    }

    private suspend fun load(text: String) {
        _state.value = try {
            val trimmed = text.trim()
            val backup = if (trimmed.startsWith(ProgressCodec.QR_PREFIX)) {
                ProgressBackup(exportedAt = clock(), books = listOf(ProgressCodec.decodeQr(trimmed)))
            } else {
                ProgressCodec.decode(trimmed)
            }
            ImportUiState.Preview(progress.preview(backup), books.books().associate { it.slug to it.title })
        } catch (e: ProgressFormatException) {
            ImportUiState.Error(e.message.orEmpty())
        }
    }

    fun confirm(mode: ImportMode) {
        val preview = (_state.value as? ImportUiState.Preview)?.preview ?: return
        viewModelScope.launch {
            _state.value = runCatching { progress.apply(preview.backup, mode) }
                .fold({ ImportUiState.Done }, { ImportUiState.Error(it.message ?: it::class.simpleName.orEmpty()) })
        }
    }

    fun dismiss() {
        _state.value = ImportUiState.Idle
    }
}

@Composable
private fun CameraPreview(onResult: (String) -> Unit, modifier: Modifier) {
    val context = LocalContext.current
    val lifecycleOwner = LocalLifecycleOwner.current
    val currentOnResult by rememberUpdatedState(onResult)
    val previewView = remember { PreviewView(context) }
    val executor = remember { Executors.newSingleThreadExecutor() }

    LaunchedEffect(lifecycleOwner) {
        val provider = ProcessCameraProvider.awaitInstance(context)
        val preview = Preview.Builder().build().also { it.setSurfaceProvider(previewView.surfaceProvider) }
        val analysis = ImageAnalysis.Builder()
            .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
            .build()
            .also { a ->
                a.setAnalyzer(executor, QrAnalyzer { text -> ContextCompat.getMainExecutor(context).execute { currentOnResult(text) } })
            }
        val selector = if (provider.hasCamera(CameraSelector.DEFAULT_BACK_CAMERA)) CameraSelector.DEFAULT_BACK_CAMERA else CameraSelector.DEFAULT_FRONT_CAMERA
        provider.unbindAll()
        provider.bindToLifecycle(lifecycleOwner, selector, preview, analysis)
    }
    DisposableEffect(Unit) {
        onDispose {
            runCatching { ProcessCameraProvider.getInstance(context).get().unbindAll() }
            executor.shutdown()
        }
    }
    AndroidView(factory = { previewView }, modifier = modifier)
}

/** Decodes the luminance (Y) plane with ZXing; no Google Play Services involved. */
private class QrAnalyzer(private val onFound: (String) -> Unit) : ImageAnalysis.Analyzer {
    private val reader = QRCodeReader()
    private val hints = mapOf(DecodeHintType.TRY_HARDER to true)
    @Volatile private var done = false

    override fun analyze(image: ImageProxy) {
        image.use {
            if (done) return
            val plane = image.planes[0]
            val buffer = plane.buffer
            val data = ByteArray(plane.rowStride * image.height)
            buffer.get(data, 0, minOf(buffer.remaining(), data.size))
            val source = PlanarYUVLuminanceSource(data, plane.rowStride, image.height, 0, 0, image.width, image.height, false)
            try {
                val text = reader.decode(BinaryBitmap(HybridBinarizer(source)), hints).text
                if (text.startsWith(ProgressCodec.QR_PREFIX)) {
                    done = true
                    onFound(text)
                }
            } catch (_: ReaderException) {
                // No QR code in this frame.
            } finally {
                reader.reset()
            }
        }
    }
}

sealed interface QrShareUiState {
    data object Loading : QrShareUiState
    data object NoProgress : QrShareUiState
    data object TooLarge : QrShareUiState
    data class Ready(val title: String, val code: ImageBitmap) : QrShareUiState
}

class QrShareViewModel(slug: String, progress: ProgressRepository, books: BookRepository) : ViewModel() {
    val state: StateFlow<QrShareUiState> = flow {
        val title = books.books().firstOrNull { it.slug == slug }?.title ?: slug
        val book = progress.exportBook(slug)
        emit(
            when {
                book == null -> QrShareUiState.NoProgress
                else -> encode(ProgressCodec.encodeQr(book))?.let { QrShareUiState.Ready(title, it) } ?: QrShareUiState.TooLarge
            },
        )
    }.flowOn(Dispatchers.Default).stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), QrShareUiState.Loading)

    private fun encode(payload: String): ImageBitmap? {
        if (payload.length > ProgressCodec.QR_MAX_LENGTH) return null
        val matrix = try {
            QRCodeWriter().encode(
                payload,
                BarcodeFormat.QR_CODE,
                0,
                0,
                mapOf(EncodeHintType.ERROR_CORRECTION to ErrorCorrectionLevel.L, EncodeHintType.MARGIN to 2),
            )
        } catch (_: WriterException) {
            return null
        }
        val pixels = IntArray(matrix.width * matrix.height) { i ->
            if (matrix[i % matrix.width, i / matrix.width]) android.graphics.Color.BLACK else android.graphics.Color.WHITE
        }
        return Bitmap.createBitmap(pixels, matrix.width, matrix.height, Bitmap.Config.ARGB_8888).asImageBitmap()
    }
}

        prepareStatement(sql).use { st ->
            args.forEachIndexed { i, a -> st.setString(i + 1, a) }
            st.executeQuery().use { rs -> buildList { while (rs.next()) add(read(rs)) } }
        }

package io.github.terlingua.books.domain

import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Test

class ProgressMergerTest {

    private fun q(id: String, correct: Boolean, selection: List<String>, at: Long) =
        QuestionProgress(id, correct, selection, selectedAt = at, updatedAt = at)

    private val local = BookProgress(
        bookSlug = "book",
        lastSectionKey = "book/a/one",
        updatedAt = 100,
        sections = listOf(
            SectionProgress("book/a/one", firstVisitedAt = 50, updatedAt = 100, questions = listOf(q("q1", true, listOf("a"), 90))),
            SectionProgress("book/a/two", firstVisitedAt = 70, updatedAt = 80),
        ),
    )

    private val remote = BookProgress(
        bookSlug = "book",
        lastSectionKey = "book/b/three",
        updatedAt = 200,
        sections = listOf(
            SectionProgress("book/a/one", firstVisitedAt = 20, updatedAt = 150, questions = listOf(q("q1", false, listOf("b"), 150), q("q2", true, listOf("c"), 140))),
            SectionProgress("book/b/three", firstVisitedAt = 190, updatedAt = 200),
        ),
    )

    @Test
    fun mergeFollowsConflictFreeRules() {
        val merged = ProgressMerger.merge(local, remote)
        val one = merged.section("book/a/one")!!
        assertEquals(20L, one.firstVisitedAt)
        assertEquals(150L, one.updatedAt)
        assertTrue("answeredCorrectly is sticky", one.question("q1")!!.answeredCorrectly)
        assertEquals("newest selection wins", listOf("b"), one.question("q1")!!.lastSelection)
        assertTrue(one.question("q2")!!.answeredCorrectly)
        assertEquals(setOf("book/a/one", "book/a/two", "book/b/three"), merged.sections.map { it.key }.toSet())
        assertEquals("book/b/three", merged.lastSectionKey)
        assertEquals(200L, merged.updatedAt)
    }

    @Test
    fun mergeIsCommutativeAndIdempotent() {
        assertEquals(ProgressMerger.merge(local, remote), ProgressMerger.merge(remote, local))
        val merged = ProgressMerger.merge(local, remote)
        assertEquals(merged, ProgressMerger.merge(merged, merged))
        assertEquals(merged, ProgressMerger.merge(merged, remote))
    }

    @Test
    fun diffCountsOnlyNewProgress() {
        val diff = ProgressMerger.diff(local, remote)
        assertEquals(1, diff.newlyVisitedSections)
        assertEquals(1, diff.newlyCorrectQuestions)
        assertEquals(2, diff.changedSelections)
        assertTrue(ProgressMerger.diff(ProgressMerger.merge(local, remote), remote).let { it.newlyCorrectQuestions == 0 && it.newlyVisitedSections == 0 })
    }

    @Test
    fun replaceDiffReportsLostAnswers() {
        val replace = ProgressMerger.diff(local, remote.copy(sections = emptyList()), ImportMode.Replace)
        assertEquals(1, replace.lostCorrectQuestions)
    }

    @Test
    fun completionRequiresAllQuestionsOrAVisit() {
        val withQuestions = SectionRef("book/a/one", "One", 1, listOf("q1", "q2"))
        val reading = SectionRef("book/a/two", "Two", 1, emptyList())
        val merged = ProgressMerger.merge(local, remote)
        assertTrue(Completion.isComplete(withQuestions, merged.section("book/a/one")))
        assertTrue(!Completion.isComplete(withQuestions, local.section("book/a/one")))
        assertTrue(Completion.isComplete(reading, local.section("book/a/two")))
        assertEquals(1f, Completion.fraction(listOf(withQuestions, reading), merged))
    }
}

==> export.sh <==
#!/usr/bin/env bash
set -euo pipefail

readonly PROJECT="terlingua-android"
readonly CONTENT_PREFIX="content/"
readonly SKIP_GLOBS=("docs/*")
FULL=0

usage() {
    cat <<USAGE
Usage: bash export.sh [-o FILE] [-f] [-- PATHSPEC...]

Writes an LLM context dump of the repository (default: docs/llm/dump.txt).

  File Tree                all git-tracked files (including ${CONTENT_PREFIX})
  Source, config, docs     full text (excluding docs/ and ${CONTENT_PREFIX})
  ${CONTENT_PREFIX}                 file tree only (excluded from dump text)
  Binary files             path only

  -o, --output FILE   write to FILE
  -f, --full          include text files inside ${CONTENT_PREFIX} too
  -h, --help          show this help

PATHSPECs are passed to git ls-files to narrow the export:

  bash export.sh -- app build-logic .github
  bash export.sh -- ':!*Test*'
  bash export.sh -f -o /tmp/book.txt -- content/books/my-book
USAGE
}

matches_any() {
    local path="$1" glob
    shift
    for glob in "$@"; do
        [[ "$path" == $glob ]] && return 0
    done
    return 1
}

is_text() {
    [[ ! -s "$1" ]] || LC_ALL=C grep -qI '' "$1"
}

classify() {
    local path="$1"
    if [[ "$path" == "$OUTPUT_REL" ]] || matches_any "$path" "${SKIP_GLOBS[@]}"; then
        echo skip
    elif [[ "$path" == "$CONTENT_PREFIX"* ]] && (( ! FULL )); then
        echo content
    elif ! is_text "$path"; then
        echo binary
    else
        echo full
    fi
}

render_tree() {
    awk -F'/' '
    {
        matched = 1
        for (i = 1; i <= NF; i++) {
            if (matched && i < NF && i < prev_len && $i == prev[i]) {
                continue
            }
            matched = 0
            indent = ""
            for (j = 1; j < i; j++) indent = indent "  "
            printf "%s%s%s\n", indent, $i, (i < NF ? "/" : "")
        }
        delete prev
        for (i = 1; i <= NF; i++) prev[i] = $i
        prev_len = NF
    }'
}

emit_file() {
    printf '==> %s <==\n' "$1"
    cat "$1"
    [[ -z "$(tail -c 1 "$1")" ]] || echo
    echo
}

write_dump() {
    echo "# ${PROJECT} export $(date -u '+%Y-%m-%dT%H:%M:%SZ')"
    if (( FULL )); then
        echo "# Full text of source, config, docs and content text files. Binary files: path only."
    else
        echo "# Full text of source, config and README. Content files: file tree only. Binary files: path only."
    fi
    echo

    echo "## File Tree (git-tracked)"
    if (( ${#TRACKED_FILES[@]} )); then
        printf '%s\n' "${TRACKED_FILES[@]}" | render_tree
    else
        echo "(none)"
    fi
    echo

    if (( ${#LISTED_FILES[@]} )); then
        echo "## Path only"
        printf '%s\n' "${LISTED_FILES[@]}"
        echo
    fi

    echo "## Files"
    echo
    if (( ${#TEXT_FILES[@]} )); then
        local path
        for path in "${TEXT_FILES[@]}"; do
            emit_file "$path"
        done
    fi
}

report() {
    local bytes
    bytes=$(wc -c < "$OUTPUT")
    {
        echo "Wrote ${OUTPUT}: ${bytes} bytes, ~$((bytes / 4)) tokens"
        echo "  git-tracked files in tree: ${#TRACKED_FILES[@]}"
        echo "  exported text files: ${#TEXT_FILES[@]}, content files (tree only): ${CONTENT_COUNT}, path only: ${#LISTED_FILES[@]}"
        if (( ${#TEXT_FILES[@]} )); then
            echo "  largest exported:"
            local path
            for path in "${TEXT_FILES[@]}"; do
                printf '%s\t%s\n' "$(wc -c < "$path")" "$path"
            done | sort -rn | head -n 10 | awk -F'\t' '{ printf "    %8d  %s\n", $1, $2 }'
        fi
    } >&2
}

main() {
    local output="" path kind
    local -a pathspecs=()
    while [[ $# -gt 0 ]]; do
        case "$1" in
            -o|--output) output="$(realpath -m "$2")"; shift 2 ;;
            -f|--full) FULL=1; shift ;;
            -h|--help) usage; exit 0 ;;
            --) shift; pathspecs=("$@"); break ;;
            *) usage >&2; exit 1 ;;
        esac
    done

    cd "$(git rev-parse --show-toplevel)"
    OUTPUT="${output:-$PWD/docs/llm/dump.txt}"
    OUTPUT_REL="$(realpath -m --relative-to="$PWD" "$OUTPUT")"

    TRACKED_FILES=()
    TEXT_FILES=()
    LISTED_FILES=()
    CONTENT_COUNT=0

    while IFS= read -r path; do
        [[ -n "$path" ]] || continue
        TRACKED_FILES+=("$path")
    done < <(git ls-files -- "${pathspecs[@]}" | LC_ALL=C sort -u)

    while IFS= read -r path; do
        [[ -f "$path" ]] || continue
        kind="$(classify "$path")"
        case "$kind" in
            skip) ;;
            content) ((CONTENT_COUNT += 1)) ;;
            binary) LISTED_FILES+=("$path") ;;
            full) TEXT_FILES+=("$path") ;;
        esac
    done < <(git ls-files --cached --others --exclude-standard -- "${pathspecs[@]}" | LC_ALL=C sort -u)

    mkdir -p "$(dirname "$OUTPUT")"
    write_dump > "$OUTPUT"
    report
}

main "$@"

##############################################################################
#
#   Gradle start up script for POSIX generated by Gradle.
#
#   Important for running:
#
#   (1) You need a POSIX-compliant shell to run this script. If your /bin/sh is
#       noncompliant, but you have some other compliant shell such as ksh or
#       bash, then to run this script, type that shell name before the whole
#       command line, like:
#
#           ksh Gradle
#
#       Busybox and similar reduced shells will NOT work, because this script
#       requires all of these POSIX shell features:
#         * functions;
#         * expansions «$var», «${var}», «${var:-default}», «${var+SET}»,
#           «${var#prefix}», «${var%suffix}», and «$( cmd )»;
#         * compound commands having a testable exit status, especially «case»;
#         * various built-in commands including «command», «set», and «ulimit».
#
#   Important for patching:
#
#   (2) This script targets any POSIX shell, so it avoids extensions provided
#       by Bash, Ksh, etc; in particular arrays are avoided.
#
#       The "traditional" practice of packing multiple parameters into a
#       space-separated string is a well documented source of bugs and security
#       problems, so this is (mostly) avoided, by progressively accumulating
#       options in "$@", and eventually passing that to Java.
#
#       Where the inherited environment variables (DEFAULT_JVM_OPTS, JAVA_OPTS,
#       and GRADLE_OPTS) rely on word-splitting, this is performed explicitly;
#       see the in-line comments for details.
#
#       There are tweaks for specific operating systems such as AIX, CygWin,
#       Darwin, MinGW, and NonStop.
#
#   (3) This script is generated from the Groovy template
#       https://github.com/gradle/gradle/blob/HEAD/platforms/jvm/plugins-application/src/main/resources/org/gradle/api/internal/plugins/unixStartScript.txt
#       within the Gradle project.
#
#       You can find Gradle at https://github.com/gradle/gradle/.
#
##############################################################################

# Attempt to set APP_HOME

# Resolve links: $0 may be a link
app_path=$0

# Need this for daisy-chained symlinks.
while
    APP_HOME=${app_path%"${app_path##*/}"}  # leaves a trailing /; empty if no leading path
    [ -h "$app_path" ]
do
    ls=$( ls -ld "$app_path" )
    link=${ls#*' -> '}
    case $link in             #(
      /*)   app_path=$link ;; #(
      *)    app_path=$APP_HOME$link ;;
    esac
done

# This is normally unused
# shellcheck disable=SC2034
APP_BASE_NAME=${0##*/}
# Discard cd standard output in case $CDPATH is set (https://github.com/gradle/gradle/issues/25036)
APP_HOME=$( cd -P "${APP_HOME:-./}" > /dev/null && printf '%s\n' "$PWD" ) || exit

# Use the maximum available, or set MAX_FD != -1 to use that value.
MAX_FD=maximum

warn () {
    echo "$*"
} >&2

die () {
    echo
    echo "$*"
    echo
    exit 1
} >&2

# OS specific support (must be 'true' or 'false').
cygwin=false
msys=false
darwin=false
nonstop=false
case "$( uname )" in                #(
  CYGWIN* )         cygwin=true  ;; #(
  Darwin* )         darwin=true  ;; #(
  MSYS* | MINGW* )  msys=true    ;; #(
  NONSTOP* )        nonstop=true ;;
esac

CLASSPATH="\\\"\\\""


# Determine the Java command to use to start the JVM.
if [ -n "$JAVA_HOME" ] ; then
    if [ -x "$JAVA_HOME/jre/sh/java" ] ; then
        # IBM's JDK on AIX uses strange locations for the executables
        JAVACMD=$JAVA_HOME/jre/sh/java
    else
        JAVACMD=$JAVA_HOME/bin/java
    fi
    if [ ! -x "$JAVACMD" ] ; then
        die "ERROR: JAVA_HOME is set to an invalid directory: $JAVA_HOME

Please set the JAVA_HOME variable in your environment to match the
location of your Java installation."
    fi
else
    JAVACMD=java
    if ! command -v java >/dev/null 2>&1
    then
        die "ERROR: JAVA_HOME is not set and no 'java' command could be found in your PATH.

Please set the JAVA_HOME variable in your environment to match the
location of your Java installation."
    fi
fi

# Increase the maximum file descriptors if we can.
if ! "$cygwin" && ! "$darwin" && ! "$nonstop" ; then
    case $MAX_FD in #(
      max*)
        # In POSIX sh, ulimit -H is undefined. That's why the result is checked to see if it worked.
        # shellcheck disable=SC2039,SC3045
        MAX_FD=$( ulimit -H -n ) ||
            warn "Could not query maximum file descriptor limit"
    esac
    case $MAX_FD in  #(
      '' | soft) :;; #(
      *)
        # In POSIX sh, ulimit -n is undefined. That's why the result is checked to see if it worked.
        # shellcheck disable=SC2039,SC3045
        ulimit -n "$MAX_FD" ||
            warn "Could not set maximum file descriptor limit to $MAX_FD"
    esac
fi

# Collect all arguments for the java command, stacking in reverse order:
#   * args from the command line
#   * the main class name
#   * -classpath
#   * -D...appname settings
#   * --module-path (only if needed)
#   * DEFAULT_JVM_OPTS, JAVA_OPTS, and GRADLE_OPTS environment variables.

# For Cygwin or MSYS, switch paths to Windows format before running java
if "$cygwin" || "$msys" ; then
    APP_HOME=$( cygpath --path --mixed "$APP_HOME" )
    CLASSPATH=$( cygpath --path --mixed "$CLASSPATH" )

    JAVACMD=$( cygpath --unix "$JAVACMD" )

    # Now convert the arguments - kludge to limit ourselves to /bin/sh
    for arg do
        if
            case $arg in                                #(
              -*)   false ;;                            # don't mess with options #(
              /?*)  t=${arg#/} t=/${t%%/*}              # looks like a POSIX filepath
                    [ -e "$t" ] ;;                      #(
              *)    false ;;
            esac
        then
            arg=$( cygpath --path --ignore --mixed "$arg" )
        fi
        # Roll the args list around exactly as many times as the number of
        # args, so each arg winds up back in the position where it started, but
        # possibly modified.
        #
        # NB: a `for` loop captures its iteration list before it begins, so
        # changing the positional parameters here affects neither the number of
        # iterations, nor the values presented in `arg`.
        shift                   # remove old arg
        set -- "$@" "$arg"      # push replacement arg
    done
fi


# Add default JVM options here. You can also use JAVA_OPTS and GRADLE_OPTS to pass JVM options to this script.
DEFAULT_JVM_OPTS='"-Xmx64m" "-Xms64m"'

# Collect all arguments for the java command:
#   * DEFAULT_JVM_OPTS, JAVA_OPTS, and optsEnvironmentVar are not allowed to contain shell fragments,
#     and any embedded shellness will be escaped.
#   * For example: A user cannot expect ${Hostname} to be expanded, as it is an environment variable and will be
#     treated as '${Hostname}' itself on the command line.

set -- \
        "-Dorg.gradle.appname=$APP_BASE_NAME" \
        -classpath "$CLASSPATH" \
        -jar "$APP_HOME/gradle/wrapper/gradle-wrapper.jar" \
        "$@"

# Stop when "xargs" is not available.
if ! command -v xargs >/dev/null 2>&1
then
    die "xargs is not available"
fi

# Use "xargs" to parse quoted args.
#
# With -n1 it outputs one arg per line, with the quotes and backslashes removed.
#
# In Bash we could simply go:
#
#   readarray ARGS < <( xargs -n1 <<<"$var" ) &&
#   set -- "${ARGS[@]}" "$@"
#
# but POSIX shell has neither arrays nor command substitution, so instead we
# post-process each arg (as a line of input to sed) to backslash-escape any
# character that might be a shell metacharacter, then use eval to reverse
# that process (while maintaining the separation between arguments), and wrap
# the whole thing up as a single "set" statement.
#
# This will of course break if any of these variables contains a newline or
# an unmatched quote.
#

eval "set -- $(
        printf '%s\n' "$DEFAULT_JVM_OPTS $JAVA_OPTS $GRADLE_OPTS" |
        xargs -n1 |
        sed ' s~[^-[:alnum:]+,./:=@_]~\\&~g; ' |
        tr '\n' ' '
    )" '"$@"'

exec "$JAVACMD" "$@"
